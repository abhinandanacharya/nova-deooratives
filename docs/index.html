Quick Start

html — your original example, working!
<script src="https://cdn.jsdelivr.net/npm/nova-deooratives/dist/mylang.umd.min.js"></script>

<script type="text/mylang">
  <for data="array(1,2,3,4)" index="default" printable="p">
    <h1>{{ p }}</h1>
  </for>
</script>

<!-- Output: 1 2 3 4 as <h1> tags -->

<let> — Variables

AttributeTypeDescription
namestringVariable name to declare
value / valexpressionValue to assign
upperflagModify the variable in the parent (enclosing) scope
mylang
<!-- Declare -->
<let name="x" value="42" />
<let name="users" value="array({name:'Alice'},{name:'Bob'})" />

<!-- Mutate counter inside a while loop -->
<let name="n" value="0" />
<while cond="n < 5">
  <p>{{ n }}</p>
  <let name="n" value="n + 1" upper />
</while>

<print> & Interpolation

Output any expression inline using {{ expr }} in text/attributes, or explicitly with <print value="expr" />. Use <log value="expr" /> for console debugging.

mylang
<let name="user" value="{ name: 'Alice', score: 95 }" />
<p>Hello, {{ user.name }}! Score: <print value="user.score + 5" /></p>
<log value="user" />

<for> — All Loop Modes

data="expr"
Iterate an array or object
of="expr"
for-of (any iterable)
in="expr"
for-in (object keys)
range="s,e,step"
Numeric range shorthand
init cond step
Classic C-style for loop
all for modes
<!-- data= mode -->
<for data="array('a','b','c')" as="v" index="i">
  <p>{{ i }}: {{ v }}</p>
</for>

<!-- range= mode -->
<for range="0,10,2" as="n">
  <span>{{ n }}</span>
</for>

<!-- Classic C-style -->
<for init="i=0" cond="i<5" step="i++">
  <p>i = {{ i }}</p>
</for>

<!-- Nested for (multiplication table) -->
<table>
  <for range="1,4" as="row">
    <tr>
      <for range="1,4" as="col">
        <td>{{ row*col }}</td>
      </for>
    </tr>
  </for>
</table>

<each> — Shorthand For-Each

Clean shorthand for iterating arrays and iterables with optional index and separator.

mylang
<each of="array('HTML', 'CSS', 'JS')" as="tech" index="i" separator=" • ">
  <span>#{{ i + 1 }} {{ tech }}</span>
</each>

<repeat> — Repeat N Times

mylang
<repeat times="5" index="i">
  <span>⭐</span>
</repeat>

<while> & <dowhile>

Use <let ... upper /> to update the loop counter in the parent scope. <dowhile> always executes its body at least once.

mylang
<let name="n" value="1" />
<while cond="n <= 16">
  <span>{{ n }} </span>
  <let name="n" value="n * 2" upper />
</while>

<let name="x" value="10" />
<dowhile cond="x < 5">
  <p>Runs once even when x={{ x }}</p>
</dowhile>

<if> / <elif> / <else>

mylang
<if cond="score >= 90">
  <p>🏆 A Grade</p>
  <elif cond="score >= 70">
    <p>✅ B Grade</p>
  </elif>
  <else>
    <p>❌ Below average</p>
  </else>
</if>
Note: <elif> and <else> are always children of <if>, not siblings. You can have unlimited <elif> branches.

<switch> / <case> / <default>

mylang
<switch value="role">
  <case match="'Admin'"><span>🛡️ Full Access</span></case>
  <case match="'Editor'"><span>✏️ Can Edit</span></case>
  <default><span>👁️ Read Only</span></default>
</switch>

<map> / <filter> / <reduce>

mylang
<!-- map -->
<map data="array(1,2,3,4,5)" as="n" separator=", ">
  <b>{{ n * n }}</b>
</map>
<!-- Output: 1, 4, 9, 16, 25 -->

<!-- filter: only even numbers -->
<filter data="range(1,20)" where="$v % 2 === 0" as="n" separator=" ">
  {{ n }}
</filter>

<!-- reduce: sum -->
<reduce data="array(10,20,30,40)" acc="sum" init="0"
        item="n" expr="sum + n" save="total" />
<p>Total = {{ total }}</p>

<try> / <catch> / <finally>

mylang
<try>
  <print value="parse(rawInput)" />
  <catch as="err"><p>Caught: {{ err }}</p></catch>
  <finally><p>Cleanup complete</p></finally>
</try>

<break /> & <continue />

mylang
<for range="1,10" as="n">
  <if cond="n === 3"><continue /></if>
  <if cond="n === 7"><break /></if>
  <span>{{ n }} </span>
</for>

Multi-Language Integrations (PHP, Python, Node, Go)

Pass arrays or objects from any backend controller into your view using data-scope='...' (inline JSON) or data-var="GLOBAL_NAME" on <script type="text/mylang">.

php — Controller → View Table
// Controller: pass $users array to view
<?php
$users = [
  ["name" => "Alice", "role" => "Admin", "active" => true],
  ["name" => "Bob",   "role" => "User",  "active" => false],
];
$json = json_encode(["users" => $users], JSON_HEX_TAG | JSON_HEX_APOS | JSON_HEX_QUOT);
?>

<!-- View: render table with nova-deooratives -->
<script src="https://cdn.jsdelivr.net/npm/nova-deooratives/dist/mylang.umd.min.js"></script>

<script type="text/mylang" data-scope='<?= $json ?>'>
  <table>
    <tr><th>#</th><th>Name</th><th>Role</th><th>Status</th></tr>
    <for of="users" as="u" index="i">
      <tr>
        <td>{{ i + 1 }}</td>
        <td>{{ u.name }}</td>
        <td>{{ u.role }}</td>
        <td><if cond="u.active">Active<else>Inactive</else></if></td>
      </tr>
    </for>
  </table>
</script>
python (Flask / Django)
# Flask / Django Controller
return render_template("users.html", payload=json.dumps({"users": users}))

<!-- View (Jinja2 / Django Template) -->
<script type="text/mylang" data-scope='{{ payload | safe }}'>
  <table>
    <for of="users" as="u">
      <tr><td>{{ u.name }}</td><td>{{ u.email }}</td></tr>
    </for>
  </table>
</script>
javascript (Node.js / Express)
// Express route passing array to HTML view
const scopeJson = JSON.stringify({ orders }).replace(/'/g, "&#39;");

<script type="text/mylang" data-scope='${scopeJson}'>
  <for of="orders" as="o">
    <p>{{ o.id }} — ${{ o.total }}</p>
  </for>
</script>
go / ruby
// Go (html/template) or Ruby (ERB): pass JSON via window variable + data-var
<script>window.PAGE_DATA = <%= @data.to_json.html_safe %>;</script>

<script type="text/mylang" data-var="PAGE_DATA">
  <for of="items" as="item">
    <li>{{ item.name }}</li>
  </for>
</script>

Security Model

⚠️ User-generated templates: Do not render untrusted user input as MyLang templates. While the evaluator blocks most attack vectors, MyLang templates should be treated like HTML — trusted content only.
  • Expressions run in strict-mode sandboxed Function
  • 20+ dangerous globals explicitly shadowed/blocked
  • No access to window, document, eval, Function
  • No fetch, XHR, localStorage, crypto, setTimeout
  • Prototype poisoning prevented via null-prototype scopes
  • Output is DOM text nodes, never innerHTML
  • Loop iteration hard-capped (default 100,000)
  • Expression compile cache bounded at 512 entries

TypeScript Support

typescript
import MyLang, {
  type RenderOptions,
  type MyLangConfig,
  type IScope,
  type MyLangError,
  createScope,
  BUILTINS,
} from 'nova-deooratives';

// Fully typed render call
const options: RenderOptions = {
  scope: { users: [{ name: 'Alice', age: 30 }] },
  maxIterations: 50_000,
  onError: (err: MyLangError) => {
    console.error(err.type, err.message, err.expression);
  },
};

MyLang.render(template, container, options);

// Typed scope manipulation
const scope: IScope = createScope(null, { x: 10, y: 20 });
scope.set('z', scope.get('x') as number + (scope.get('y') as number));

// Typed evaluate
const result: unknown = MyLang.evaluate('x * 2', { x: 21 });
console.log(result); // 42

Real-World: To-Do List

Track tasks with conditional status icons and a live completion counter via <reduce>.

html
<let name="todos" value="array(
  {text:'Buy groceries', done:true},
  {text:'Write docs', done:false},
  {text:'Ship v1.0', done:false}
)" />
<ul>
  <each of="todos" as="todo" index="i">
    <li>
      <if cond="todo.done">
        ✅ <s>{{ todo.text }}</s>
        <else>⬜ {{ todo.text }}</else>
      </if>
    </li>
  </each>
</ul>
<reduce data="todos" acc="done" init="0"
        item="t" expr="done + (t.done ? 1 : 0)" save="doneCount" />
<p><b>{{ doneCount }} / {{ len(todos) }}</b> complete</p>

Real-World: Controller → View Data Table

Render a team directory table from a controller array (works identically with PHP json_encode, Python json.dumps, or Node.js).

html

  {name:'Bob', role:'Developer', salary:82000, active:true},
  {name:'Carol', role:'Designer', salary:70000, active:false}

      <td>{{ u.role }}</td>
      <td>

      </td>

  </for>
</table>
<p style="margin-top:8px">Total Payroll: <b>${{ payroll }}</b></p>

Real-World: E-Commerce Invoice

Compute line totals, filter in-stock items, and calculate grand total using <for> + <reduce>.

html
<let name="cart" value="array(
  {item:'Mechanical Keyboard', qty:1, price:89},
  {item:'USB-C Hub', qty:2, price:25},
  {item:'Laptop Stand', qty:1, price:45}
)" />
<reduce data="cart" acc="t" init="0" item="c" expr="t + c.qty * c.price" save="subtotal" />
<table>
  <tr><th>Item</th><th>Qty</th><th>Total</th></tr>
  <for of="cart" as="c">
    <tr>
      <td>{{ c.item }}</td>
      <td>{{ c.qty }}</td>
      <td>${{ c.qty * c.price }}</td>
    </tr>
  </for>
</table>
<p style="margin-top:8px">Subtotal: <b>${{ subtotal }}</b>
  <if cond="subtotal >= 150"> (🎉 Free Shipping!)</if>
</p>
